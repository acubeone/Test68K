; TCS -> Test CaSe
; file.tcs
; little-endian
; 4-bytes alignment

file-format:
	[header]
	[env]
	[entry0]
	[blob0]
	[entry1]
	[blob1]
	...
	[entryN]
	[blobN]


[header - 32-bytes]
header + 0x00 -> MAGIC_WORD : u8[4]
	content: 0x54 0x36 0x38 0x4b ; ASCII: T68K
header + 0x04 -> VERSION : u8
	content: 0x02 ; Version 2
header + 0x05 -> CPU_MODEL : u8
	content: 0=68000, 1=68010
header + 0x06 -> RESERVED : u8[10]
	content: 0x00 0x00 0x00 0x00 ... 0x00
header + 0x10 -> SEED : u64
	content: Random seed used for test generation
header + 0x18 -> ENTRY_COUNT : u32
	content: 0xffff_ffff is unknown/streaming
header + 0x1c -> RESERVED : u8[4]
	content: 0x00 0x00 0x00 0x00

[env - variable]
env + 0x00 -> RAM_LEN : u16
	content: TCS only uses 64KB RAM region padded to 4-bytes boundary
env + 0x02 -> RESERVED : u8[2]
	content: 0x00 0x00
env + 0x04 -> RAM_BASE : u32
	content: Base address of RAM. Most significant byte must be 0x00
env + 0x08 -> RAM_CONTENT : variable
	content: u8[RAM_LEN] ; It should start at RAM_BASE address

[entry - variable]
entry + 0x00 -> MAGIC_WORD : u8[4]
	content: 0x54 0x43 0x53 0x32 ; ASCII: TCS2
entry + 0x04 -> ENTRY_LEN : u32
	content: blob bytes length (includes padding)
entry + 0x08 -> BLOB : variable
	content: [blob]

[blob - ordered]
	name_len : u32
	name : u8[name_len] ; ASCII non-null terminated string
	padding : u8[] ; align to 4-bytes boundary

	op_word_count : u32
	op_words : u16[op_word_count]
	padding : u8[] ; align to 4-bytes boundary

	pre_regs : u32[23] ; [D0..A7, PC, SR, USP, SSP, VBR, SFC, DFC]
	pre_mem_len : u32
	pre_mem : [ram][pre_mem_len] ; Opcode write diff + optional entropy

	post_regs : u32[23] ; [D0..A7, PC, SR, USP, SSP, VBR, SFC, DFC]
	post_mem_len : u32
	post_mem : [mem][post_mem_len] ; Diff from pre_ram

	mem_op_len : u32
	mem_ops : [mem_op][mem_op_len]

	cycles : u32
	exception_vector : u16 ; vector index, 0 if none
	exception_kind : u16 ; 0=ok, 1=bus_err, 2=addr_err, 3=illegal,
						 ; 4=div_by_zero, 5=chk, 6=trapv, 7=privilege,
						 ; 8=trace, 9=linea, 10=linef, 11=format_err,
						 ; 12=irq_uninit, 13=irq_spurious, 14=trap, 15=bkpt

[mem - ordered]
	addr : u32
	byte : u8
	padding: u8[3] = 0x00 0x00 0x00

[mem_op - ordered]
	addr : u32
	data : u16
	flags : u8 ; bit0: (0=read, 1=write); bit1: (0=byte, 1=word); bit5-7: 3-bit FC
	padding : u8 = 0x00
